import DeliveryTimeModel from '../models/deliveryTime';
import model from '../models/branch';
import orderModel from '../models/order';

async function ckeckOfflinePOSmercadopago() {
  try {
    let extraData = [];
    const dateNow = new Date();
    const dateMinuteLess = new Date(dateNow.getTime() - (1 * 60 * 1000)); // 1 minute less

    let branchesToClose = await model.find({
      platforms: {
        $elemMatch: {
          platform: new mongoose.Types.ObjectId(mercadopagoId),
          isActive: true
        }
      },
      lastGetNews: { $lt: dateMinuteLess },
      alreadyClosedMercadoPago: { $in: [null, false] }
    }).populate("chain");
    let branchesToOpen = await model.find({
      platforms: {
        $elemMatch: {
          platform: new mongoose.Types.ObjectId(mercadopagoId),
          isActive: true
        }
      },
      lastGetNews: { $gt: dateMinuteLess },
      alreadyClosedMercadoPago: { $in: [null, true] }
    }).populate("chain");

    const tokenAPICloud = await Token.getToken();
    if (!tokenAPICloud) {
      Log.saveError(null, { message: 'SmartCloud Communication Error: MercadoPago -C Failed to get token from Cloud', platform: 'MercadoPago' });
      throw new Error('Failed to get token from Cloud');
    }
     
    for(let tenant of tenants){
      console.log(`for of tenat: ${tenant.name}`);
      let branchesToClose2 = branchesToClose.filter( r => r.chain?.chain.trim().toLowerCase() === tenant.name.trim().toLocaleLowerCase());
      let branchesToOpen2  = branchesToOpen.filter( r => r.chain?.chain.trim().toLowerCase() === tenant.name.trim().toLocaleLowerCase());

      if (branchesToClose2.length === 0 && branchesToOpen2.length === 0) {
        console.log('Continue tenant', tenant.name, 'matches close', branchesToClose2.length, 'open', branchesToOpen2.length);
        continue;
      }

      const closePromises = branchesToClose2.map(async (branch) => {
        const branchId = branch.branchId;
        const branchData = { branchId, results: [] };
        const branchReference = branch.platforms.find(pm => pm.platform == mercadopagoId).branchReference;
        const branchReferences = Array.isArray(branchReference) ? branchReference : [branchReference];
   
        const mercadoPagoPlatform = branch.platforms?.find(p => p.platform?.toString() === mercadopagoId?.toString());
 
        let user_id = mercadoPagoPlatform?.mercadoPagoUserId;
        const headersToken = {
          headers: {
            'Authorization': `Bearer ${tokenAPICloud}`,
            'TraceKey': '1234',
            'TenantId': tenant.tenantId
          }
        }
 
        let urlTokenMercadoPagoCloud = `${mercadoPagoGetToken_uri}${user_id}`;    
        let tokenMercadoPagoCloud;
        let decryptedTokenMercadoPagoCloud;
        try {
          tokenMercadoPagoCloud = await axios.get(urlTokenMercadoPagoCloud, headersToken);
          if (tokenMercadoPagoCloud?.status === 204) {
            Log.saveError(null, { 
              message: `SmartCloud Communication Error: MercadoPago -C No Content (204) fetching token from Cloud [branchId: ${branchId}] [user_id: ${user_id}]`,  
              error: {
                message: 'No Content (204)',
                customDataAfter: {
                  action: 'getCloudToken',
                  platformId: 8,
                  branch: {
                    branchId: branchId,
                    user_id: user_id
                  },
                  request: {
                    url: urlTokenMercadoPagoCloud,
                    method: 'GET'
                  },
                  response: {
                    status: 204,
                    data: null
                  }
                }
              },
              platform: 'MercadoPago' 
            });            
          }
          decryptedTokenMercadoPagoCloud = await Token.decrypt(tokenMercadoPagoCloud?.data?.token);
        } catch (err) {
          Log.saveError(err, { 
            message: `SmartCloud Communication Error: MercadoPago -C Error fetching token from Cloud [branchId: ${branchId}] [user_id: ${user_id}]`,  
            error: {
              message: err?.message,
              customDataAfter: {
                action: 'getCloudToken',
                platformId: 8,
                branch: {
                  branchId: branchId,
                  user_id: user_id
                },
                request: {
                  url: urlTokenMercadoPagoCloud,
                  method: 'GET'
                },
                response: {
                  status: err?.response?.status,
                  data: err?.response?.data
                }
              }
            },
            platform: 'MercadoPago' 
          });
        }       
        const headersConfig = {
          headers: {
            'Authorization': `Bearer ${decryptedTokenMercadoPagoCloud}`
          }
        }

        await Promise.allSettled(branchReferences.map(async (store_id) => {
          let body = {
            "status": "paused"
          };
   
          const urlAvailability = `${baseUrlMercadoPago}/proximity-integration/stores/${store_id}/status`;
   
          try {
            await axios.put(urlAvailability, body, headersConfig);
            branchData.results.push({
              branchReference: store_id,
              type: "Closed",
              result: true
            });
          } catch (error) {
            console.log(error);
            branchData.results.push({
              branchReference: store_id,
              type: "Closed",
              result: false
            });
            Log.saveError(error, { 
            message: `Branch OpenClose Error: MercadoPago checkOfflinePOSmercadopago CLOSE ${branchId}`, 
            action: 'checkOfflinePOSmercadopago',
            platformId: 8,
            operation: 'CLOSE',
            branch: {
                branchReferenceId: store_id,
                user_id: user_id,
                branchId: branchId
            },
            request: {
                url: urlAvailability,
                method: 'POST',
                body: body,
                header: headersConfig
            },
            response: {
                status: error?.response?.status,
                data: error?.response?.data
            }
            });
   
          }
        }));
        await model.findByIdAndUpdate(branch.id, { alreadyClosedMercadoPago: true });
        console.log('alreadyClosedMercadoPago: true updated');
        extraData.push(branchData);
      });
   
      const openPromises = branchesToOpen2.map(async (branch) => {
        const branchId = branch.branchId;
        const branchData = { branchId, results: [] };
        const branchReference = branch.platforms.find(pm => pm.platform == mercadopagoId).branchReference;
        const branchReferences = Array.isArray(branchReference) ? branchReference : [branchReference];
        const mercadoPagoPlatform = branch.platforms?.find(p => p.platform?.toString() === mercadopagoId?.toString());    
        let user_id = mercadoPagoPlatform?.mercadoPagoUserId;
        const headersToken = {
          headers: {
            'Authorization': `Bearer ${tokenAPICloud}`,
            'TraceKey': '1234',
            'TenantId': tenant.tenantId
          }
        }
        let urlTokenMercadoPagoCloud = `${mercadoPagoGetToken_uri}${user_id}`;    
        let tokenMercadoPagoCloud;
        let decryptedTokenMercadoPagoCloud;
        try {
          tokenMercadoPagoCloud = await axios.get(urlTokenMercadoPagoCloud, headersToken);
          if (tokenMercadoPagoCloud?.status === 204) {
            Log.saveError(null, { 
              message: `SmartCloud Communication Error: MercadoPago -C No Content (204) fetching token from Cloud [branchId: ${branchId}] [user_id: ${user_id}]`,  
              error: {
                message: 'No Content (204)',
                customDataAfter: {
                  action: 'getCloudToken',
                  platformId: 8,
                  branch: {
                    branchId: branchId,
                    user_id: user_id
                  },
                  request: {
                    url: urlTokenMercadoPagoCloud,
                    method: 'GET'
                  },
                  response: {
                    status: 204,
                    data: null
                  }
                }
              },
              platform: 'MercadoPago' 
            });            
          }
          decryptedTokenMercadoPagoCloud = await Token.decrypt(tokenMercadoPagoCloud?.data?.token);
        } catch (err) {
          Log.saveError(err, { 
            message: `SmartCloud Communication Error: MercadoPago -C Error fetching token from Cloud [branchId: ${branchId}] [user_id: ${user_id}]`,  
            error: {
              message: err?.message,
              customDataAfter: {
                action: 'getCloudToken',
                platformId: 8,
                branch: {
                  branchId: branchId,
                  user_id: user_id
                },
                request: {
                  url: urlTokenMercadoPagoCloud,
                  method: 'GET'
                },
                response: {
                  status: err?.response?.status,
                  data: err?.response?.data
                }
              }
            },
            platform: 'MercadoPago' 
          });
        }
     
        if (!decryptedTokenMercadoPagoCloud) {
          Log.saveError(null, { message: `Platform Validation Error: MercadoPago -C Failed to decrypt MercadoPago token from Cloud [branchId: ${branchId}] [user_id: ${user_id}]`, platform: 'MercadoPago' });
        }
     
        const headersConfig = {
          headers: {
            'Authorization': `Bearer ${decryptedTokenMercadoPagoCloud}`
          }
        }

        await Promise.allSettled(branchReferences.map(async (store_id) => {
          let body = {
            "status": "enabled"
          };
   
          const urlAvailability = `${baseUrlMercadoPago}/proximity-integration/stores/${store_id}/status`;
          try {
            await axios.put(urlAvailability, body, headersConfig);  
            branchData.results.push({
              branchReference: store_id,
              type: "Open",
              result: true
            });
          } catch (error) {
            branchData.results.push({
              branchReference: store_id,
              type: "Open",
              result: false
            });
            Log.saveError(error, { 
            message: `Branch OpenClose Error: MercadoPago checkOfflinePOSmercadopago OPEN ${branchId}`, 
            action: 'checkOfflinePOSmercadopago',
            platformId: 8,
            operation: 'OPEN',
            branch: {
                branchReferenceId: store_id,
                user_id: user_id,
                branchId: branchId
            },
            request: {
                url: urlAvailability,
                method: 'POST',
                body: body,
                header: headersConfig
            },
            response: {
                status: error?.response?.status,
                data: error?.response?.data
            }
            });
          }
        }));
        await model.findByIdAndUpdate(branch.id, { alreadyClosedMercadoPago: false });
        extraData.push(branchData);
      });
