Resumen General de la Versión

Este release del Concentrador añade soporte backend para la exportación de sucursales a formato de hoja de cálculo. La inclusión de controladores y rutas para listar y contar sucursales activas filtrando por países y cadenas permite realizar descargas masivas y parciales de los datos de sucursales configurados.

const getBranchesToExport = async (req, res) => {
  try {   
    const { chain, country, page, size } = req.body;
    let matchStage = { deletedAt: null };
    if (country?.length) matchStage['address.country'] = { $in: country };
    if (chain?.length) matchStage.chain = { $in: chain };
    const branchesToExport = await model.find(matchStage)
      .select('branchId platforms chain address.country name')
      .sort({ branchId: 1 })
      .skip(page * size)
      .limit(size)
      .lean();
      return res.status(200).json(branchesToExport).end();
  } catch (error) {
    Log.saveError(error, { message: 'Falló getBranchesToExport concentrador', platform: 'Concentrador' });
  }
};
