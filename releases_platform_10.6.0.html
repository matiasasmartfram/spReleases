Notas de Release: Platform 10.6.0

Este release de Platform introduce una reforma de logging integral (Reforma LogErros) diseñada para estandarizar la clasificación, categorización y seguridad de los registros de error en todo el flujo de integración de la aplicación. Se añade soporte para salida estructurada por stdout en formato JSON, sanitización recursiva de cabeceras sensibles en peticiones HTTP para evitar fugas de credenciales en la base de datos, y se categorizan todos los logs bajo las áreas de Integración, Órdenes, Infraestructura y Performance.

Resumen General de la Versión

Este release introduce la Reforma de LogErros en Platform, unificando, estandarizando y categorizando de forma estructurada los errores del sistema bajo cuatro grandes categorías: Integración, Órdenes, Infraestructura y Performance. Como parte de la reforma, se introduce la capacidad de publicar logs en consola en formato JSON (stdout) y la sanitización recursiva de cabeceras de red para eliminar credenciales, tokens y firmas criptográficas sensibles antes de persistir la información en MongoDB.

'use strict';

function isSensitiveHeaderKey(key) {
    if (!key || typeof key !== 'string') return false;
    const normalized = key.toLowerCase();
    return normalized === 'authorization' ||
        normalized.includes('authorization') ||
        normalized.includes('signature') ||
        normalized.endsWith('sign');
}

function isHeadersContainerKey(key) {
    if (!key || typeof key !== 'string') return false;
    const normalized = key.toLowerCase();
    return normalized === 'headers' ||
        normalized === 'header' ||
        normalized.includes('headers') ||
        normalized.includes('header');
}

function sanitizeHeadersObject(headers) {
    if (!headers || typeof headers !== 'object') return headers;
    if (Array.isArray(headers)) return headers.map(sanitizeHeadersObject);

    const sanitized = {};

    Object.entries(headers).forEach(([key, value]) => {
        if (isSensitiveHeaderKey(key)) {
            sanitized[key] = '[REDACTED]';
        } else if (value && typeof value === 'object') {
            sanitized[key] = sanitizeHeadersObject(value);
        } else {
            sanitized[key] = value;
        }
    });

    return sanitized;
}

function sanitizeHeaders(data) {
    // Valido si es un Object o Array para sanitizar
    if (!data || typeof data !== 'object') return data;
    if (Array.isArray(data)) return data.map(sanitizeHeaders);

    const sanitized = {};

    Object.entries(data).forEach(([key, value]) => {
        if (isHeadersContainerKey(key) && value && typeof value === 'object') {
            sanitized[key] = sanitizeHeadersObject(value);
        } else if (value && typeof value === 'object') {
            sanitized[key] = sanitizeHeaders(value);
        } else {
            sanitized[key] = value;
        }
    });

    return sanitized;
}
